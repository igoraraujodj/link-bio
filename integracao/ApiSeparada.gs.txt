// Planilha "Finanças da Casa" — a mesma que o bot do Telegram usa.
const SHEET_ID = '1jfM5O9wRqrnonUyxHiTstgF5Jp2t6R4jkhH8Ng6vZBQ';

// Senha do endereço. Serve só pra ninguém abrir por acaso.
const CHAVE = 'casa-ccvdngz86y';


// ── Daqui pra baixo não precisa mexer ───────────────────────────────────

const FUSO = 'America/Sao_Paulo';

const ABA_LANC  = 'lancamentos';
const ABA_FIXOS = 'fixos';
const ABA_ORC   = 'orcamento';

const CATEGORIAS = [
  'Mercado', 'Alimentação', 'Delivery', 'Transporte', 'Casa', 'Contas',
  'Saúde', 'Farmácia', 'Educação', 'Lazer', 'Vestuário', 'Assinaturas',
  'Pets', 'Presentes', 'Outros'
];

// Posições fixas, iguais às que o bot grava.
const COL = {
  id: 1, registrado: 2, data: 3, valor: 4, categoria: 5, descricao: 6,
  estabelecimento: 7, metodo: 8, quem: 9, origem: 10, confianca: 11,
  status: 12, link: 13
};

function doGet(e) {
  if (!e || !e.parameter || e.parameter.k !== CHAVE) {
    return ContentService.createTextOutput('<div>Link inválido ou expirado.</div>')
                         .setMimeType(ContentService.MimeType.TEXT);
  }

  // ContentService e não HtmlService: só ele responde com o cabeçalho que
  // o navegador exige para ler de outro endereço.
  return ContentService
    .createTextOutput(JSON.stringify(dadosDoApp_()))
    .setMimeType(ContentService.MimeType.JSON);
}

function dadosDoApp_() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  return {
    lancamentos: lancamentos_(ss),
    orcamento: orcamento_(ss),
    fixos: fixos_(ss),
    categorias: CATEGORIAS,
    hoje: Utilities.formatDate(new Date(), FUSO, 'yyyy-MM-dd')
  };
}

/**
 * Os últimos 13 meses de lançamentos ativos.
 *
 * O rateio é procurado PELO NOME da coluna, não por posição. A planilha já
 * tem `parcelas` e `parcela` nas colunas 14 e 15: fixar a posição faria o
 * script ler o número de parcelas achando que era rateio, e o acerto do mês
 * sairia errado — errado com cara de certo, que é o pior tipo.
 */
function lancamentos_(ss) {
  const sh = ss.getSheetByName(ABA_LANC);
  if (!sh || sh.getLastRow() < 2) return [];

  const largura = sh.getLastColumn();
  const cabecalho = sh.getRange(1, 1, 1, largura).getValues()[0]
                      .map(function (t) { return String(t || '').trim().toLowerCase(); });
  const colRateio = cabecalho.indexOf('rateio');   // -1 enquanto não existir

  const corte = new Date();
  corte.setMonth(corte.getMonth() - 13);

  const dados = sh.getRange(2, 1, sh.getLastRow() - 1, largura).getValues();
  const saida = [];

  dados.forEach(function (r) {
    if (r[COL.status - 1] !== 'ativo') return;
    const data = r[COL.data - 1];
    if (!(data instanceof Date) || data < corte) return;

    saida.push({
      d: Utilities.formatDate(data, FUSO, 'yyyy-MM-dd'),
      v: Number(r[COL.valor - 1]) || 0,
      c: r[COL.categoria - 1] || 'Outros',
      t: String(r[COL.descricao - 1] || r[COL.estabelecimento - 1] || ''),
      e: String(r[COL.estabelecimento - 1] || ''),
      m: String(r[COL.metodo - 1] || ''),
      q: String(r[COL.quem - 1] || ''),
      l: String(r[COL.link - 1] || ''),
      r: colRateio >= 0 ? String(r[colRateio] || '') : ''
    });
  });

  saida.sort(function (a, b) { return a.d < b.d ? 1 : a.d > b.d ? -1 : 0; });
  return saida;
}

/** Teto por categoria, da aba `orcamento`. */
function orcamento_(ss) {
  const sh = ss.getSheetByName(ABA_ORC);
  if (!sh || sh.getLastRow() < 2) return {};
  const mapa = {};
  sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues().forEach(function (r) {
    if (r[0] && Number(r[1]) > 0) mapa[r[0]] = Number(r[1]);
  });
  return mapa;
}

/** Contas fixas e renda, da aba `fixos`. */
function fixos_(ss) {
  const sh = ss.getSheetByName(ABA_FIXOS);
  const vazio = { contas: [], renda: 0, fixo: 0 };
  if (!sh || sh.getLastRow() < 2) return vazio;

  const out = { contas: [], renda: 0, fixo: 0 };
  sh.getRange(2, 1, sh.getLastRow() - 1, 4).getValues().forEach(function (r) {
    const nome = String(r[0] || '').trim();
    const valor = Number(r[3]) || 0;
    if (!nome || !valor) return;

    if (String(r[1] || '').toLowerCase() === 'renda') { out.renda += valor; return; }

    out.contas.push({
      nome: nome,
      grupo: String(r[1] || 'Vida'),
      categoria: String(r[2] || 'Outros'),
      valor: valor
    });
    out.fixo += valor;
  });

  out.contas.sort(function (a, b) { return b.valor - a.valor; });
  return out;
}

/**
 * Rode esta função uma vez antes de implantar (▶ Executar).
 * Se aparecerem números, está tudo certo.
 */
function testar() {
  const d = dadosDoApp_();
  Logger.log('Lançamentos dos últimos 13 meses: ' + d.lancamentos.length);
  Logger.log('Categorias com teto de orçamento: ' + Object.keys(d.orcamento).length);
  Logger.log('Contas fixas: ' + d.fixos.contas.length);
  Logger.log('Renda mensal somada: ' + d.fixos.renda);
  Logger.log('Contas fixas somadas: ' + d.fixos.fixo);
  if (d.lancamentos.length) {
    Logger.log('Lançamento mais recente: ' + JSON.stringify(d.lancamentos[0]));
    Logger.log('Coluna de rateio: ' + (d.lancamentos[0].r ? 'encontrada' : 'ainda não existe (tudo bem)'));
  }
}
